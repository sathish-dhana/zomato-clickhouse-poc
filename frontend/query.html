Example Queries
Count logs by environment
SELECT env, count() as count
FROM foo_service
GROUP BY env
ORDER BY count DESC
Find error logs
SELECT ts, env, container_id, msg
FROM foo_service
WHERE msg LIKE '%error%'
ORDER BY ts DESC
LIMIT 100
Logs per hour in the last day
SELECT 
    toStartOfHour(ts) as hour,
    count() as count
FROM foo_service
WHERE ts >= now() - INTERVAL 1 DAY
GROUP BY hour
ORDER BY hour
Filter by specific container
SELECT ts, env, msg, _others['user_id'] as user_id
FROM foo_service
WHERE container_id = 'container_1'
ORDER BY ts DESC
LIMIT 50
